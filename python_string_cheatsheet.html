Most Frequent Character
def most_frequent_char(s):
  if not s: return None
  freq = {}
  for ch in s:
    freq[ch] = freq.get(ch, 0) + 1
  max_char = max(freq, key=freq.get)
  return max_char, freq[max_char]

most_frequent_char("programming")
# → ('g', 2)
dict.get(key, default) avoids KeyError on first occurrence

Check Duplicate Characters
def has_duplicates(s):
  return len(s) != len(set(s))

has_duplicates("hello")  # True
has_duplicates("world")  # False
def has_duplicates(s):
  seen = set()
  for ch in s:
    if ch in seen: return True
    seen.add(ch)
  return False

has_duplicates("hello")  # True
def has_duplicates(s):
  freq = {}
  for ch in s:
    freq[ch] = freq.get(ch, 0) + 1
    if freq[ch] > 1: return True
  return False

Anagram Check
def is_anagram(s1, s2):
  return sorted(s1) == sorted(s2)

# O(n log n) — sort both, compare
is_anagram("listen", "silent")
# → True

# O(n) alternative:
from collections import Counter
Counter(s1) == Counter(s2)
Counter approach is faster for large strings

Reverse a String
def reverse_string(s):
  return s[::-1]     # slice trick

reverse_string("hello") # "olleh"

# Alternatives:
"".join(reversed(s))   # iterator
list(s)[::-1]           # list then join
s[::-1] is idiomatic and fastest in CPython

First Non-Repeating Character
from collections import Counter

def first_unique_char(s):
  count = Counter(s)
  for ch in s:
    if count[ch] == 1:
      return ch
  return None

first_unique_char("aabbcde")
# → 'c'
Two-pass O(n): count all, then scan order

Palindrome Check
def is_palindrome(s):
  s = s.lower().replace(" ", "")
  return s == s[::-1]

is_palindrome("madam")      # True
is_palindrome("Race car")   # True
is_palindrome("hello")      # False
Normalize: lowercase + strip spaces before comparing

Remove Duplicates (preserve order)
def remove_duplicates(s):
  return "".join(dict.fromkeys(s))

remove_duplicates("programming")
# → "progamin"

# dict.fromkeys preserves insertion order
# (Python 3.7+). set() does NOT preserve.
dict.fromkeys(s) → unique keys in original order

Longest Unique Substring (Sliding Window)
def longest_unique_substring(s):
  seen = set()
  left = max_len = 0
  for right in range(len(s)):
    while s[right] in seen:
      seen.remove(s[left])
      left += 1
    seen.add(s[right])
    max_len = max(max_len, right-left+1)
  return max_len

longest_unique_substring("abcabcbb")
# → 3  ("abc")
Two-pointer O(n) — shrink window when duplicate found

Longest Common Substring — Brute Force vs DP
def longest_common_substring(s1, s2):
  max_sub = ""
  for i in range(len(s1)):
    for j in range(i+1, len(s1)+1):
      sub = s1[i:j]
      if sub in s2 and len(sub) > len(max_sub):
        max_sub = sub
  return max_sub

longest_common_substring("abcdxyz", "xyzabcd")
# → "abcd"
def longest_common_substring(s1, s2):
  m, n = len(s1), len(s2)
  dp = [[0]*(n+1) for _ in range(m+1)]
  max_len = end_idx = 0

  for i in range(1, m+1):
    for j in range(1, n+1):
      if s1[i-1] == s2[j-1]:
        dp[i][j] = dp[i-1][j-1] + 1
        if dp[i][j] > max_len:
          max_len = dp[i][j]
          end_idx = i

  return s1[end_idx - max_len : end_idx]

longest_common_substring("abcdxyz", "xyzabcd")
# → "abcd"
How the DP table works — "abcd" vs "xyzabcd"
dp[i][j] = length of longest common substring ending at s1[i-1] and s2[j-1]
      x  y  z  a  b  c  d
   [  0  0  0  0  0  0  0  0 ]
a  [  0  0  0  0 1  0  0  0 ]
b  [  0  0  0  0  0 2  0  0 ]
c  [  0  0  0  0  0  0 3  0 ]
d  [  0  0  0  0  0  0  0 4 ]  ← max_len=4
x  [  0  1  0  0  0  0  0  0 ]
y  [  0  0  2  0  0  0  0  0 ]
z  [  0  0  0  3  0  0  0  0 ]
Rule: if chars match → dp[i][j] = dp[i-1][j-1] + 1 (extend diagonal)
if no match → dp[i][j] = 0 (reset, substring must be contiguous)
Track max value + its position → slice s1 to recover the substring

Functional Tools — map · filter · reduce · input
map — transform each element
# Square every number
a = [1, 2, 3, 4]
a = list(map(lambda x: x*x, a))
# → [1, 4, 9, 16]

# Convert types
nums = list(map(int, ["1","2","3"]))
# → [1, 2, 3]
filter — keep elements matching condition
# Keep values > 2
a = [1, 2, 3, 4]
a = list(filter(lambda x: x > 2, a))
# → [3, 4]

# Keep only even numbers
a = list(filter(lambda x: x%2==0, a))
# → [4]
reduce — fold to single value
from functools import reduce

b = [1, 2, 3, 4]
total = reduce(lambda x,y: x+y, b)
# → 10  (1+2+3+4)

product = reduce(lambda x,y: x*y, b)
# → 24  (1*2*3*4)
input parsing — split + cast
# Space-separated ints from stdin
a = list(map(int, input().split()))
# "1 2 3 4"  → [1, 2, 3, 4]

# As strings
a = list(map(str, input().split()))
# "a b c"  → ['a', 'b', 'c']

# Single line of multiple values
x, y, z = map(int, input().split())
